  ```cpp

  {
      damage += 2;
      std::cout << "Security Drone uses Pulse Scan.\n";
  }
  if (enemy.getName() == "Corporate Enforcer" && (std::rand() % 100) < 25)
  {
      damage += 3;
      std::cout << "Corporate Enforcer uses Armor Breaker.\n";
  }
  ```
